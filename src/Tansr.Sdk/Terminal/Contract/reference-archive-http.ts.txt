const errorMap = {
  invalid_request: [400, 'none'], protocol_version_mismatch: [400, 'none'], unauthorized: [401, 'none'],
  forbidden: [403, 'none'], unsupported_capability: [422, 'none'], stale_generation: [409, 'none'],
  archive_gap: [409, 'none'], source_unavailable: [503, 'query-status'], invalid_coverage: [422, 'none'],
  capacity_exceeded: [429, 'same-request'], request_id_conflict: [409, 'none'], request_expired: [410, 'none'],
  receipt_expired: [410, 'none'], payload_too_large: [413, 'none'], material_rejected: [422, 'none'],
  binding_closed: [409, 'none'], internal_error: [500, 'query-status'], epoch_unavailable: [503, 'same-request'],
  binding_conflict: [409, 'none'],
  stream_cursor_unknown: [409, 'none'], stream_cursor_expired: [410, 'none'], stream_gap: [409, 'none'],
} as const;
type WireErrorCode = keyof typeof errorMap;
/** 包内扩展路由复用同一错误状态与重试合同；不增加公共协议出口。 */
export function archiveHttpErrorPolicy(code: WireErrorCode): (typeof errorMap)[WireErrorCode] {
  return errorMap[code];
}
export class ArchiveHttpError extends Error {
  constructor(readonly code: WireErrorCode) { super(code); this.name = 'ArchiveHttpError'; }
}
function fail(code: WireErrorCode): never { throw new ArchiveHttpError(code); }
function parse<T>(schema: z.ZodType<T>, value: unknown): T {
  const result = schema.safeParse(value); if (!result.success) fail('invalid_request'); return result.data;
}
function output<T>(schema: z.ZodType<T>, value: unknown, maximum: number): T {
  try {
    // 输出先走原纯数据编码；不允许 getter/toJSON 或失去原数字约束后伪装合法结果。
    const plain = decodeArchiveMetadata(encodeArchiveMetadata(value, { maxBytes: maximum }));
    const result = schema.safeParse(plain); if (result.success) return result.data;
  } catch { /* 仅回固定内部错误，不暴露宿主路径或异常文本。 */ }
  return fail('internal_error');
}
function recoveryOutput(value: unknown, maximum: number): Sdk2AckRebaseReceipt {
  try {
    if (!value || typeof value !== 'object' || types.isProxy(value) || Array.isArray(value) ||
      ![null, Object.prototype].includes(Object.getPrototypeOf(value))) fail('internal_error');
    const descriptors = Object.getOwnPropertyDescriptors(value), plain: Record<string, unknown> = {};
    for (const name of Reflect.ownKeys(descriptors)) {
      if (typeof name !== 'string') fail('internal_error');
      const field = descriptors[name]!; if (!field.enumerable || !Object.hasOwn(field, 'value')) fail('internal_error');
      // 每个旧ACK/receipt保持原控制帽；仅恢复外包拥有独立总字节帽。
      plain[name] = decodeArchiveMetadata(encodeArchiveMetadata(field.value));
    }
    serialize(plain, maximum); return parse(archiveRecoverySchema('AckRebaseReceipt'), plain) as Sdk2AckRebaseReceipt;
  } catch { return fail('internal_error'); }
}
function translated(error: unknown): WireErrorCode {
  if (error instanceof ArchiveHttpError || error instanceof ArchiveHttpEventError) return error.code;
  // 原事件库错误保留类身份；任意同名 code 不得成为可透传的 HTTP 诊断。
  if (error instanceof EventStoreError) {
    switch (error.code) {
      case 'stream_cursor_unknown': case 'stream_cursor_expired': case 'stream_gap': case 'capacity_exceeded': return error.code;
      case 'unavailable': case 'reconciliation_required': return 'source_unavailable';
      default: return 'internal_error';
    }
  }
  // B 的必需副本尚未耐久是来源暂不可用，不能伪报内部故障或自动重发新 ACK。
  if (error instanceof ArchiveSourceTransferError && error.code === 'dependency_held') return 'source_unavailable';
  if (error instanceof SqliteSpoolError || error instanceof MaterialSupplyError ||
    error instanceof ArchiveRecordJournalError || error instanceof ArchiveSourceTransferError) {
    switch (error.code) {
      case 'invalid_input': return 'invalid_request';
      case 'capacity_exceeded': return 'capacity_exceeded';
      case 'operation_conflict': case 'entry_conflict': return 'request_id_conflict';
      case 'stale_generation': return 'stale_generation';
      case 'epoch_expired': return 'epoch_unavailable';
      case 'request_expired': return 'request_expired';
      case 'binding_closed': return 'binding_closed';
      case 'revision_conflict': case 'head_conflict': return 'binding_conflict';
      case 'archive_gap': return 'archive_gap';
      case 'artifact_unavailable': case 'source_unavailable': case 'entry_released': return 'source_unavailable';
      case 'unsupported_capability': return 'unsupported_capability';
      default: return 'internal_error';
    }
  }
  return 'internal_error';
}

/** 页与下载块有独立较大的响应帽；只编码可信输出，不改原正文（正文仍base64原字节）。 */
function serialize(value: unknown, maxBytes: number): Buffer {
  const data = JSON.stringify(value, (_key, item: unknown) => {
    if (typeof item === 'number' && (!Number.isSafeInteger(item) || item < 0 || Object.is(item, -0))) throw new TypeError('Invalid response integer');
    if (item && typeof item === 'object' && !Array.isArray(item)) {
      return Object.fromEntries(Object.entries(item).sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0));
    }
    return item;
  });
  if (data === undefined) throw new TypeError('Missing response');
  const bytes = Buffer.from(data); if (bytes.length > maxBytes) fail('capacity_exceeded'); return bytes;
}
function reply(res: ServerResponse, status: number, value: unknown, maxBytes = CONTROL_BYTES): void {
  if (res.destroyed || res.writableEnded) return;
  const bytes = serialize(value, maxBytes);
  res.writeHead(status, { 'content-type': 'application/json; charset=utf-8', 'content-length': bytes.length, 'cache-control': 'no-store' });
  res.end(bytes);
}
function component(value: string): string {
  try { return decodeURIComponent(value); } catch { return fail('invalid_request'); }
}
function query(raw: string, required: readonly string[], optional: readonly string[] = []): Record<string, string> {
  const result: Record<string, string> = Object.create(null);
  for (const pair of raw.split('&')) {
    const equal = pair.indexOf('=');
    if (equal < 1 || pair.includes('+')) fail('invalid_request');
    const key = pair.slice(0, equal), value = component(pair.slice(equal + 1));
    if (![...required, ...optional].includes(key) || Object.hasOwn(result, key) || value === '') fail('invalid_request');
    result[key] = value;
  }
  if (required.some(key => !Object.hasOwn(result, key))) fail('invalid_request');
  if (result.protocol !== PROTOCOL) fail('protocol_version_mismatch');
  return result;
}
function count(value: string, min: number, max: number): number {
  if (!/^(0|[1-9][0-9]*)$/.test(value)) fail('invalid_request');
  return parse(integer(min, max), Number(value));
}
function readGenerations(params: Record<string, string>): ArchiveGenerations {
  return parse(generations, { historyEpoch: params.historyEpoch, deletionGeneration: params.deletionGeneration, projectionRevision: params.projectionRevision });
}
function sameGenerations(left: ArchiveGenerations, right: ArchiveGenerations): boolean {
  return left.historyEpoch === right.historyEpoch && left.deletionGeneration === right.deletionGeneration && left.projectionRevision === right.projectionRevision;
}
async function readBody(req: IncomingMessage, maximum: number, recovery = false): Promise<unknown> {
  const media = req.headers['content-type'] ?? '';
  if (!/^application\/json(?:\s*;\s*charset\s*=\s*(?:utf-8|"utf-8"))?\s*$/i.test(media) ||
    (req.headers['content-encoding'] !== undefined && req.headers['content-encoding'].toLowerCase() !== 'identity')) fail('invalid_request');
  const chunks: Buffer[] = []; let bytes = 0;
  for await (const part of req) {
    const chunk = Buffer.isBuffer(part) ? part : Buffer.from(part as Uint8Array); bytes += chunk.length;
    if (bytes > maximum) fail('payload_too_large'); chunks.push(chunk);
  }
  let result: unknown;
  try {
    const raw = Buffer.concat(chunks);
    if (recovery) {
      const text = new TextDecoder('utf-8', { fatal: true, ignoreBOM: true }).decode(raw); result = JSON.parse(text);
      // 恢复扩展明确使用规范控制JSON，比较原字节拒绝重复键/替代数字/BOM。
      if (!serialize(result, maximum).equals(raw)) fail('invalid_request');
    } else result = decodeArchiveMetadata(raw, { maxBytes: maximum });
  } catch { return fail('invalid_request'); }
  if (!result || typeof result !== 'object' || Array.isArray(result)) fail('invalid_request');
  if ((result as { protocol?: unknown }).protocol !== PROTOCOL) fail('protocol_version_mismatch');
  return result;
}

export interface ArchiveHttpHost {
  readonly applicationScopeId: string;

export function createArchiveHttpRoutes(options: { host: ArchiveHttpHost; admission: AdmissionController;
  authenticate: RequestAuthenticator['authenticate']; track(work: Promise<void>): void }) {
  const maxControl = options.host.controlBytes ?? CONTROL_BYTES;
  parse(integer(1024, CONTROL_BYTES), maxControl); parse(id, options.host.applicationScopeId);
  let inflight = 0;
  return (req: IncomingMessage, res: ServerResponse): boolean => {
    const raw = req.url ?? '';
    if (!raw.startsWith('/v3/sdk2/bindings') && !raw.startsWith('/v3/sdk2/operations') &&
      !raw.startsWith('/v3/sdk2/capabilities') && !raw.startsWith('/v3/sdk2/sessions/')) return false;
    let occupied = false, reserved = false, reservedBytes = maxControl, requestId: string = randomUUID();
    const work = (async () => {
      if (inflight >= 128 || options.admission.overloaded()) fail('capacity_exceeded');
      inflight++; occupied = true;
      if (Buffer.byteLength(raw) > 8192) fail('payload_too_large');
      if (raw.includes('#')) fail('invalid_request');
      const question = raw.indexOf('?'), path = question < 0 ? raw : raw.slice(0, question), rawQuery = question < 0 ? '' : raw.slice(question + 1);
      const segments = path.split('/').slice(3).map(component);
      const method = req.method ?? 'GET';
      if (method !== 'GET' && method !== 'POST') fail('invalid_request');
      if (method === 'GET' && (req.headers['transfer-encoding'] !== undefined || (req.headers['content-length'] !== undefined && req.headers['content-length'] !== '0'))) fail('invalid_request');
      if (method === 'POST' && question >= 0) fail('invalid_request');
      const recovery = method === 'POST' && segments.length === 4 && segments[0] === 'bindings' && segments[2] === 'archive' && segments[3] === 'ack-rebases';
      if (recovery)
        reservedBytes += SDK2_ACK_RECOVERY_LIMITS.requestEnvelopeBytes;
      if (method === 'POST') {
        if (!options.admission.reserveBody(reservedBytes).admitted) fail('capacity_exceeded'); reserved = true;
      }
      const body = method === 'POST' ? await readBody(req, reservedBytes, recovery) : undefined;
      const authenticate = async (expectedUser?: string) => {
        let authenticated: Awaited<ReturnType<AgentSessionsOptions['authenticate']>>;
        try { authenticated = await options.authenticate(req, res); } catch (error) {
          if (error instanceof AuthenticationUnavailableError) return fail(error.code === 'overloaded' ? 'capacity_exceeded' : 'source_unavailable');
          return fail('unauthorized');
        }
        if (!authenticated?.endUserId || (expectedUser !== undefined && authenticated.endUserId !== expectedUser)) fail('unauthorized');
        if (req.aborted || res.destroyed || res.writableEnded) fail('unauthorized');
        return authenticated.endUserId;
      };
      const admit = async () => {
        const user = await authenticate();
        if (!(await options.admission.rateLimit(req, user)).admitted) fail('capacity_exceeded');
        // rateLimit也是异步宿主边界；在任何资源/业务访问之前复核当前身份。
        await authenticate(user); return user;
      };
      if (segments.length === 1 && segments[0] === 'capabilities' && method === 'GET') {
        if (question >= 0) fail('invalid_request');
        const user = await admit(), read = options.host.capabilities;
        if (!read) fail('unsupported_capability');
        const value = await read.call(options.host, user); await authenticate(user);
        if (options.host.capabilities !== read) fail('forbidden');
        reply(res, 200, output(discovery, value, maxControl), maxControl); return;
      }
      const sessionTarget = segments.length === 3 && segments[0] === 'sessions' && segments[2] === 'binding-target' && method === 'GET';
      const creating = segments.length === 1 && segments[0] === 'bindings' && method === 'POST';
      let creationQuery: Record<string, string> | undefined;
      if (segments.length === 1 && segments[0] === 'operations' && method === 'GET') {
        const candidate = query(rawQuery, ['protocol', 'operation', 'operationEpoch', 'requestId'], ['bindingId', 'sessionId']);
        if (candidate.operation === 'binding-create') {
          if (candidate.bindingId !== undefined || candidate.sessionId === undefined) fail('invalid_request');
          creationQuery = candidate;
        } else if (candidate.sessionId !== undefined || candidate.bindingId === undefined) fail('invalid_request');
      }
      if (sessionTarget || creating || creationQuery) {
        let sessionId: string, createInput: ArchiveBindingCreateRequest | undefined, queryInput: ArchiveBindingIdentity | undefined;
        if (sessionTarget) { query(rawQuery, ['protocol']); sessionId = parse(legacyId, segments[1]); }
        else if (creating) { createInput = parse(createBinding, body); sessionId = createInput.target.sessionId; requestId = createInput.request.requestId; }
        else {
          sessionId = parse(legacyId, creationQuery!.sessionId);
          queryInput = parse(identity, { operationEpoch: creationQuery!.operationEpoch, requestId: creationQuery!.requestId }); requestId = queryInput.requestId;
        }
        const user = await admit(), resolveSession = options.host.resolveSession;
        if (!resolveSession) fail('unsupported_capability');
        const port = resolveSession.call(options.host, user, sessionId); if (!port) fail('forbidden');
        const reauthorize = async () => {
          await authenticate(user);
          if (options.host.resolveSession !== resolveSession || resolveSession.call(options.host, user, sessionId) !== port) fail('forbidden');
        };
        if (sessionTarget) {
          const read = port.readTarget, value = await read.call(port); await reauthorize();
          if (port.readTarget !== read) fail('forbidden');
          const view = output(bindingTarget, value, maxControl);
          if (view.target.sessionId !== sessionId) fail('forbidden');
          reply(res, 200, view, maxControl);
        } else if (createInput) {
          const create = port.create; if (!create) fail('unsupported_capability');
          const value = await create.call(port, createInput); await reauthorize();
          if (port.create !== create) fail('forbidden');
          const view = output(bindingView, value, maxControl);
          if (view.target.sessionId !== sessionId || view.scope.applicationScopeId !== options.host.applicationScopeId || view.scope.endUserId !== user) fail('forbidden');
          reply(res, 201, view, maxControl);
        } else {
          const read = port.queryCreate; if (!read) fail('unsupported_capability');
          const value = await read.call(port, queryInput!); await reauthorize();
          if (port.queryCreate !== read) fail('forbidden');
          if (value === null) fail('receipt_expired');
          const receipt = output(mutationReceipt, value, maxControl);
          if (receipt.operation !== 'binding-create' || receipt.request.operationEpoch !== queryInput!.operationEpoch || receipt.request.requestId !== queryInput!.requestId) fail('internal_error');
          reply(res, 200, receipt, maxControl);
        }
        return;
      }
      const generationKeys = ['historyEpoch', 'deletionGeneration', 'projectionRevision'];
      let bindingId: string;
      let operation: 'records' | 'artifact' | 'ack' | 'ack-rebase' | 'upload' | 'upload-status' | 'response' | 'status' | 'operation' | 'events' | 'view' | 'archive-status' | 'close';
      let lastEventId: string | undefined;
      let params: Record<string, string> = Object.create(null);
      let input: unknown = body;
      if (segments.length === 1 && segments[0] === 'operations' && method === 'GET') {
        params = query(rawQuery, ['protocol', 'operation', 'bindingId', 'operationEpoch', 'requestId']);
        bindingId = parse(id, params.bindingId); parse(identity, { operationEpoch: params.operationEpoch, requestId: params.requestId });
        if (!['archive-ack', 'material-response', 'binding-close'].includes(params.operation!)) fail('unsupported_capability');
        requestId = params.requestId!; operation = 'operation';
      } else {
        if (segments[0] !== 'bindings') fail('invalid_request'); bindingId = parse(id, segments[1]);
        if (segments.length === 2 && method === 'GET') { query(rawQuery, ['protocol']); operation = 'view'; }
        else if (segments.length === 4 && segments[2] === 'archive' && segments[3] === 'status' && method === 'GET') {
          query(rawQuery, ['protocol']); operation = 'archive-status';
        } else if (segments.length === 3 && segments[2] === 'close' && method === 'POST') {
          input = parse(closeBinding, body); operation = 'close';
        } else if (segments.length === 3 && segments[2] === 'events' && method === 'GET') {
          params = query(rawQuery, ['protocol']);
          const values: string[] = [];
          for (let index = 0; index < req.rawHeaders.length; index += 2) {
            if (req.rawHeaders[index]!.toLowerCase() === 'last-event-id') values.push(req.rawHeaders[index + 1]!);
          }
          if (values.length > 1) fail('invalid_request');
          if (values.length === 1) lastEventId = parse(id, values[0]);
          operation = 'events';
        } else if (segments.length === 4 && segments[2] === 'archive' && segments[3] === 'records' && method === 'GET') {
          params = query(rawQuery, ['protocol', ...generationKeys, 'limit', 'maxBytes'], ['afterSequence']);
          input = { generations: readGenerations(params), afterSequence: params.afterSequence === undefined ? null : parse(seq, params.afterSequence),
            limit: count(params.limit!, 1, 128), maxBytes: count(params.maxBytes!, 1024, 1048576) }; operation = 'records';
        } else if (segments.length === 5 && segments[2] === 'archive' && segments[3] === 'artifacts' && method === 'GET') {
          params = query(rawQuery, ['protocol', ...generationKeys, 'offset', 'maxBytes']);
          input = { generations: readGenerations(params), artifactId: parse(id, segments[4]), offset: count(params.offset!, 0, 33554431), maxBytes: count(params.maxBytes!, 1, 262144) }; operation = 'artifact';
        } else if (segments.length === 4 && segments[2] === 'archive' && segments[3] === 'acks' && method === 'POST') {
          input = parse(ack, body); operation = 'ack';
        } else if (segments.length === 4 && segments[2] === 'archive' && segments[3] === 'ack-rebases' && method === 'POST') {
          input = parse(archiveRecoverySchema('AckRebaseRequest'), body); operation = 'ack-rebase';
        } else if (segments.length === 3 && segments[2] === 'material-responses' && method === 'POST') {
          input = parse(response, body); operation = 'response';
        } else if (segments[2] === 'materials' && (segments.length === 4 || segments.length === 6 || segments.length === 7)) {
          parse(id, segments[3]); if (segments.length > 4) { if (segments[4] !== 'uploads') fail('invalid_request'); parse(id, segments[5]); }
          if (segments.length === 7 && segments[6] === 'chunks' && method === 'POST') { input = parse(upload, body); operation = 'upload'; }
          else if (method === 'GET' && segments.length !== 7) { query(rawQuery, ['protocol']); operation = segments.length === 4 ? 'status' : 'upload-status'; }
          else return fail('invalid_request');
        } else return fail('unsupported_capability');
      }
      if (method === 'POST') {
        const item = input as { bindingId: string; materialRequestId?: string; artifactId?: string; request?: { requestId: string } };
        if (item.bindingId !== bindingId || (operation === 'upload' && (item.materialRequestId !== segments[3] || item.artifactId !== segments[5]))) fail('invalid_request');
        if (item.request) requestId = item.request.requestId;
      }
      if (operation === 'upload') {
        const chunk = input as z.infer<typeof upload>; const bytes = Buffer.from(chunk.base64, 'base64');
        // 此处只判断入站字节；后续介质损坏仍保留未知/需对账，不能一概冒充客户端错误。
        if (bytes.length !== chunk.bytes || bytes.toString('base64') !== chunk.base64 || hashArchiveRaw(bytes) !== chunk.chunkSha256) fail('invalid_request');
      }
      const identityValue = { endUserId: await admit() };
      const ports = options.host.resolve(identityValue.endUserId, bindingId); if (!ports) fail('forbidden');
      const control = ports.control.read();
      if (!control || control.registration.bindingId !== bindingId || control.registration.scope.endUserId !== identityValue.endUserId ||
        control.registration.scope.applicationScopeId !== options.host.applicationScopeId) fail('forbidden');
      if (control.state === 'unbound') fail('binding_closed');
      if (operation === 'view' || operation === 'archive-status' || operation === 'close' || operation === 'ack-rebase') {
        const reauthorize = async () => {
          await authenticate(identityValue.endUserId);
          if (options.host.resolve(identityValue.endUserId, bindingId) !== ports) fail('forbidden');
          const current = ports.control.read();
          if (!current || current.registration.bindingId !== bindingId || current.registration.scope.endUserId !== identityValue.endUserId ||
            current.registration.scope.applicationScopeId !== options.host.applicationScopeId) fail('forbidden');
        };
        if (operation === 'view') {
          const read = ports.readView; if (!read) fail('unsupported_capability');
          const value = await read.call(ports); await reauthorize(); if (ports.readView !== read) fail('forbidden');
          const view = output(bindingView, value, maxControl);
          if (view.bindingId !== bindingId || view.scope.applicationScopeId !== options.host.applicationScopeId || view.scope.endUserId !== identityValue.endUserId ||
            view.target.sessionId !== control.registration.sessionId) fail('forbidden');
          reply(res, 200, view, maxControl);
        } else if (operation === 'archive-status') {
          if (!control.acceptedCapabilities.includes('archive-transfer-v1')) fail('unsupported_capability');
          const read = ports.readArchiveStatus; if (!read) fail('unsupported_capability');
          const value = await read.call(ports); await reauthorize(); if (ports.readArchiveStatus !== read) fail('forbidden');
          const status = output(archiveStatus, value, maxControl);
          if (status.bindingId !== bindingId || status.sourceId !== control.registration.sourceId || status.sourceGeneration !== control.registration.sourceGeneration) fail('forbidden');
          reply(res, 200, status, maxControl);
        } else if (operation === 'ack-rebase') {
          if (!control.acceptedCapabilities.includes('archive-transfer-v1')) fail('unsupported_capability');
          const recover = ports.rebaseAck; if (!recover) fail('unsupported_capability');
          const request = input as Sdk2AckRebaseRequest;
          const value = await recover.call(ports, request); await reauthorize(); if (ports.rebaseAck !== recover) fail('forbidden');
          const maximum = 2 * maxControl + SDK2_ACK_RECOVERY_LIMITS.responseEnvelopeBytes;
          const result = recoveryOutput(value, maximum);
          const equal = (left: unknown, right: unknown) => encodeArchiveMetadata(left).toString() === encodeArchiveMetadata(right).toString();
          if (result.bindingId !== bindingId || !equal(result.previous, request.previous) || !equal(result.request, request.request) ||
            !equal(result.next.request, request.request) || BigInt(result.next.expectedRevision) <= BigInt(request.previous.expectedRevision) ||
            !equal({ ...result.next, request: request.previous.request, expectedRevision: request.previous.expectedRevision }, request.previous) ||
            result.receipt.operation !== 'archive-ack' || !equal(result.receipt.request, result.next.request) || result.receipt.bindingId !== bindingId) fail('internal_error');
          reply(res, 200, result, maximum);
        } else {
          const close = ports.close; if (!close) fail('unsupported_capability');
          const closeInput = input as ArchiveBindingCloseRequest;
          const value = await close.call(ports, closeInput); await reauthorize(); if (ports.close !== close) fail('forbidden');
          const receipt = output(mutationReceipt, value, maxControl);
          if (receipt.bindingId !== bindingId || receipt.operation !== 'binding-close' || receipt.request.operationEpoch !== closeInput.request.operationEpoch || receipt.request.requestId !== closeInput.request.requestId) fail('internal_error');
          reply(res, 200, receipt, maxControl);
        }
        return;
      }
      if (operation === 'events') {
        const port = ports.events; if (!port) fail('unsupported_capability');
        const endUserId = identityValue.endUserId;
        if (req.aborted || res.destroyed || res.writableEnded) return;
        if (!options.admission.tryAdmitSse(endUserId).admitted) fail('capacity_exceeded');
        res.once('close', () => options.admission.releaseSse(endUserId));
        req.socket.setTimeout(0);
        const authorize = async () => {
          let currentIdentity: Awaited<ReturnType<AgentSessionsOptions['authenticate']>>;
          try { currentIdentity = await options.authenticate(req, res); } catch (error) {
            if (error instanceof AuthenticationUnavailableError) return fail(error.code === 'overloaded' ? 'capacity_exceeded' : 'source_unavailable');
            return fail('unauthorized');
          }
          if (!currentIdentity?.endUserId || currentIdentity.endUserId !== endUserId) fail('unauthorized');
          const currentPorts = options.host.resolve(endUserId, bindingId);
          if (!currentPorts || currentPorts.events !== port || currentPorts.control !== ports.control) fail('forbidden');
          const current = currentPorts.control.read();
          if (!current || current.registration.bindingId !== bindingId || current.registration.scope.endUserId !== endUserId ||
            current.registration.scope.applicationScopeId !== options.host.applicationScopeId) fail('forbidden');
          if (current.state === 'unbound') fail('binding_closed');
          // 绑定/代际/当前应用权限及安装事实的细核仍由原宿主events端口持有，不由路由重造authority。
        };
        await sendArchiveHttpEvents({ req, res, port, scope: { applicationScopeId: options.host.applicationScopeId, endUserId, bindingId },
          ...(lastEventId === undefined ? {} : { lastEventId }), authorize });
        return;
      }
      const materials = (): SqliteMaterialSupply => {
        const supply = ports.materials;
        if (!control.acceptedCapabilities.includes('context-materials-v1') || !supply) fail('unsupported_capability');
        if (supply.bindingId !== bindingId || !control.registration.relatedBindingIds.includes(supply.storageBindingId)) fail('forbidden');
        const owner = ports.spool.bindingIdentity(supply.storageBindingId);
        if (owner.applicationScopeId !== options.host.applicationScopeId || owner.endUserId !== identityValue.endUserId) fail('forbidden');
        return supply;
      };
      if (['records', 'artifact', 'ack'].includes(operation) && !control.acceptedCapabilities.includes('archive-transfer-v1')) fail('unsupported_capability');
      let result: unknown; let status = 200; let responseMax = maxControl;
      if (operation === 'records') {
        const request = input as Parameters<SqliteArchiveRecordJournal['list']>[0];
        if (!sameGenerations(request.generations, control.target.generations)) fail('stale_generation');
        result = ports.journal.list(request); responseMax = request.maxBytes;
      } else if (operation === 'artifact') {
        const request = input as Parameters<SqliteArchiveRecordJournal['readArtifact']>[0];
        if (!sameGenerations(request.generations, control.target.generations)) fail('stale_generation');
        const chunk = ports.journal.readArtifact(request);
        result = { protocol: PROTOCOL, bindingId, artifactId: chunk.artifact.artifactId, sourceId: chunk.artifact.sourceId,
          generations: request.generations, offset: chunk.offset, bytes: chunk.body.byteLength, totalBytes: chunk.artifact.bytes,
          sha256: chunk.artifact.sha256, chunkSha256: chunk.chunkSha256, base64: Buffer.from(chunk.body).toString('base64') };
        responseMax = ARTIFACT_RESPONSE_BYTES;
      } else if (operation === 'ack') result = ports.acknowledger.acknowledge(input as z.infer<typeof ack>);
      else if (operation === 'upload') result = materials().uploadChunk(input as z.infer<typeof upload>);
      else if (operation === 'response') { result = materials().respond(input as z.infer<typeof response>); status = 202; }
      else if (operation === 'status') result = materials().status(segments[3]!);
      else if (operation === 'upload-status') result = materials().uploadStatus(segments[3]!, segments[5]!);
      else {
        const request = { operationEpoch: params.operationEpoch!, requestId: params.requestId! };
        result = params.operation === 'material-response' ? materials().queryResponse(request)
          : ports.control.query(params.operation as 'archive-ack' | 'binding-close', request)?.receipt;
        if (!result) fail('receipt_expired');
      }
      if (!result || typeof result !== 'object' || (result as { bindingId?: string }).bindingId !== bindingId) throw new Error('Mismatched host binding');
      reply(res, status, result, responseMax);
    })().catch(error => {
      if (res.headersSent || res.destroyed || res.writableEnded) return;
      const code = translated(error), [status, retryAction] = errorMap[code];
      if (status === 429 || status === 503) res.setHeader('retry-after', '5');
      reply(res, status, { protocol: PROTOCOL, requestId, code, status, message: code, retryAction });
    }).finally(() => { if (occupied) inflight--; if (reserved) options.admission.releaseBody(reservedBytes); });
    options.track(work); return true;
  };
}
