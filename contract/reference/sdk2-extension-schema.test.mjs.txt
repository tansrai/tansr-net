// This small reference model expresses normative cross-request rules JSON Schema cannot enforce.
// It is deliberately not imported by product code and is not a claim of durable implementation.
function canonical(value) {
  if (value === null || typeof value !== 'object') return JSON.stringify(value);
  if (Array.isArray(value)) return '[' + value.map(canonical).join(',') + ']';
  return '{' + Object.keys(value).sort().map((key) => JSON.stringify(key) + ':' + canonical(value[key])).join(',') + '}';
}

// Whole requested intervals are checked, including overlap with already acknowledged records.
// Artifact identity is immutable across categories and records; receipt SHA is raw bytes SHA.
function receiptSets(records, sourceId) {
  const objects = new Map(); const payloads = new Map(); const attachments = new Map();
  const add = (ref, category) => {
    if (ref.sourceId !== sourceId) throw new Error('invalid_coverage');
    const prior = objects.get(ref.artifactId);
    if (prior && canonical(prior) !== canonical(ref)) throw new Error('invalid_coverage');
    objects.set(ref.artifactId, ref);
    if (!category.has(ref.artifactId)) category.set(ref.artifactId, { artifactId: ref.artifactId, sha256: ref.sha256, state: 'durably-stored' });
  };
  for (const record of records) {
    add(record.payload, payloads);
    for (const ref of record.attachments) add(ref, attachments);
  }
  return { payloads: [...payloads.values()], attachments: [...attachments.values()], physicalObjects: objects.size };
}

function checkWholeAck(body, records, controlBytes) {
  if (!validator('ArchiveAckRequest')(body)) throw new Error('invalid_coverage');
  if (Buffer.byteLength(canonical(body), 'utf8') > controlBytes) throw new Error('payload_too_large');
  if (records.length < 1 || records.length > 128) throw new Error('invalid_coverage');
  const expected = receiptSets(records, body.sourceId);
  if (canonical(body.payloads) !== canonical(expected.payloads) || canonical(body.attachments) !== canonical(expected.attachments)) throw new Error('invalid_coverage');
  return expected;
}

class BindingContractModel {
  state = 'active';
  revision = 0n;
  acked = 0n;
  published = 3n;
  generation = copy(generations);
  pending = 3;
  authorized = true;
  retired = false;
  nowMs = 0;
  epochExpiresAtMs = 3600000;
  terminalReceiptRetentionMs = 604800000;
  receipts = new Map();
  controlBytes = limits.controlBytes;
  archiveAckFormat = ackFormat;
  hostRecords = new Map(['1', '2', '3'].map((sequence) => [sequence, { ...copy(record), sequence }]));
  legacyHistory = Object.freeze(['current-effective-user', 'current-effective-assistant']);
  hostHeads = new Map([['1', digest], ['2', hash('head-2')], ['3', hash('head-3')]]);

  key(body, operation) {
    return canonical(['app-1', 'alice', body.bindingId, operation, body.request.operationEpoch, body.request.requestId]);
  }

  query(body, operation) {
    if (!this.authorized) throw new Error('forbidden');
    const previous = this.receipts.get(this.key(body, operation));
    if (!previous) throw new Error('receipt_expired');
    return copy(previous.result);
  }

  collectTerminalReceipts() {
    if (this.nowMs < this.epochExpiresAtMs) throw new Error('active_epoch_receipts_must_remain');
    for (const [key, entry] of this.receipts) {
      if (!entry.pending && this.nowMs >= entry.completedAtMs + this.terminalReceiptRetentionMs) this.receipts.delete(key);
    }
  }

  mutate(body, operation) {
    if (!this.authorized) throw new Error('forbidden');
    const semantic = copy(body); delete semantic.request;
    const witness = domainHash('tansr.sdk2.operation.v1', canonical({ scope: ['app-1', 'alice'], operation, semantic }));
    const key = this.key(body, operation);
    const previous = this.receipts.get(key);
    if (previous) {
      if (previous.witness !== witness) throw new Error('request_id_conflict');
      return copy(previous.result);
    }
    if (this.retired || this.nowMs >= this.epochExpiresAtMs || body.request.operationEpoch !== request.operationEpoch) throw new Error('receipt_expired');
    if (canonical(body.generations) !== canonical(this.generation)) throw new Error('stale_generation');
    if (body.expectedRevision !== this.revision.toString()) throw new Error('binding_conflict');
    if (this.state === 'closed') throw new Error('binding_closed');
    if (operation === 'archive-ack') {
      if (body.sourceId !== 'source-1' || body.sourceGeneration !== 'source-boot-1') throw new Error('forbidden');
      const from = BigInt(body.coverage.fromSequence); const through = BigInt(body.coverage.throughSequence);
      if (from < 1n || from > through || from > this.acked + 1n || through < this.acked || through > this.published ||
        this.hostHeads.get(body.coverage.throughSequence) !== body.coverage.headDigest) throw new Error('invalid_coverage');
      if (through - from + 1n > 128n || body.ackFormat !== this.archiveAckFormat) throw new Error('invalid_coverage');
      const records = [];
      for (let sequence = from; sequence <= through; sequence++) {
        const record = this.hostRecords.get(sequence.toString());
        if (!record || record.sequence !== sequence.toString() || canonical(record.target.generations) !== canonical(this.generation)) throw new Error('invalid_coverage');
        records.push(record);
      }
      checkWholeAck(body, records, this.controlBytes);
      this.acked = through; this.pending = Number(this.published - through);
      if (this.state === 'closing' && this.pending === 0) this.state = 'closed';
    } else if (operation === 'binding-close') this.state = this.pending === 0 ? 'closed' : 'closing';
    else throw new Error('unsupported_capability');
    this.revision++;
    const result = { revision: this.revision.toString(), state: 'completed', acknowledged: this.acked.toString() };
    this.receipts.set(key, { witness, result, completedAtMs: this.nowMs, pending: false });
    return copy(result);
  }
}

test('SDK2 split ACK: discovery and binding explicitly select the supported format only for archive', () => {
  const discovery = copy(goldens.CapabilitiesResponse);
  const missing = copy(discovery); delete missing.archiveAckFormats;
  invalid('CapabilitiesResponse', missing);
  invalid('CapabilitiesResponse', { ...discovery, archiveAckFormats: [] });
  invalid('CapabilitiesResponse', { ...discovery, archiveAckFormats: ['mixed-receipts-v0'] });
  const materialsOnly = { ...discovery, capabilities: ['context-materials-v1'], archiveAckFormats: [] };
  assert.equal(validator('CapabilitiesResponse')(materialsOnly), true);
  invalid('CapabilitiesResponse', { ...materialsOnly, archiveAckFormats: [ackFormat] });
  invalid('BindingView', { ...binding, archiveAckFormat: null });
  invalid('BindingView', { ...binding, archiveAckFormat: 'mixed-receipts-v0' });
  const materialsBinding = { ...binding, acceptedCapabilities: ['context-materials-v1'], archiveAckFormat: null };
  assert.equal(validator('BindingView')(materialsBinding), true);
  invalid('BindingView', { ...materialsBinding, archiveAckFormat: ackFormat });
  const legacyCreate = copy(create); delete legacyCreate.archive.ackFormat;
  invalid('BindingCreateRequest', legacyCreate);
  // Optional archive rejection must preserve a separately required available capability.
  const negotiate = (input, offered) => {
    const supported = (capability) => offered.capabilities.includes(capability) &&
      (capability !== 'archive-transfer-v1' || offered.archiveAckFormats.includes(input.archive.ackFormat));
    if (input.requiredCapabilities.some((value) => !supported(value))) throw new Error('unsupported_capability');
    const accepted = [...input.requiredCapabilities, ...input.optionalCapabilities.filter(supported)];
    return { accepted, format: accepted.includes('archive-transfer-v1') ? input.archive.ackFormat : null };
  };
  assert.throws(() => negotiate(create, materialsOnly), /unsupported_capability/);
  const optional = { ...create, requiredCapabilities: ['context-materials-v1'], optionalCapabilities: ['archive-transfer-v1'] };
  assert.deepEqual(negotiate(optional, materialsOnly), { accepted: ['context-materials-v1'], format: null });
  assert.deepEqual(negotiate(optional, discovery), { accepted: ['context-materials-v1', 'archive-transfer-v1'], format: ackFormat });
});

function recordWithAttachments(count, sequence = '1') {
  return { ...copy(record), sequence, recordId: 'record-' + sequence,
    payload: { ...artifact, artifactId: 'payload-' + sequence },
    attachments: Array.from({ length: count }, (_, index) => ({ ...artifact, artifactId: 'attachment-' + sequence + '-' + index })) };
}

function ackFor(records, overrides = {}) {
  const sets = receiptSets(records, 'source-1');
  return { ...copy(ack), payloads: sets.payloads, attachments: sets.attachments,
    coverage: { fromSequence: records[0].sequence, throughSequence: records.at(-1).sequence,
      headDigest: records.at(-1).sequence === '1' ? digest : hash('head-' + records.at(-1).sequence) }, ...overrides };
}

// Executable reference semantics only: the product must supply durable transactions and core leases.
class MaterialContractModel {
  state = 'pending';
  nowMs = 0;
  authorized = true;
  retired = false;
  epochExpiresAtMs = 3600000;
  bindingState = 'active';
  bindingRevision = 0n;
  revision = 0n;
  coreHolds = 0;
  coreQueued = false;
  acknowledged = false;
  receivedBytes = 0;
  uploads = new Map();
  receipts = new Map();

  constructor(saved = materialRequest, quota = limits.materialBytes, issuedAtMs = 0) {
    this.saved = copy(saved);
    this.currentTarget = copy(saved.target);
    this.currentSourceId = saved.sourceId;
    this.currentSourceGeneration = saved.sourceGeneration;
    this.nowMs = issuedAtMs;
    this.deadlineMs = issuedAtMs + saved.remainingTtlMs;
    this.chunkBytes = saved.chunkBytes;
    assert.ok(Number.isInteger(this.chunkBytes) && this.chunkBytes > 0 && this.chunkBytes <= limits.materialChunkBytes);
    this.witnesses = new Map();
    for (const item of saved.requestedRecords) for (const witness of [item.payload, ...item.attachments]) {
      const previous = this.witnesses.get(witness.artifactId);
      if (previous && canonical(previous) !== canonical(witness)) throw new Error('material_rejected');
      this.witnesses.set(witness.artifactId, copy(witness));
      if (Math.ceil(witness.bytes / this.chunkBytes) > 16) throw new Error('capacity_exceeded');
    }
    this.reservedBytes = [...this.witnesses.values()].reduce((sum, item) => sum + item.bytes, 0);
    if (this.reservedBytes > Math.min(saved.maxBytes, limits.materialBytes, quota)) throw new Error('capacity_exceeded');
  }

  authorize(body) {
    if (!this.authorized || body.bindingId !== this.saved.bindingId || body.materialRequestId !== this.saved.materialRequestId) throw new Error('forbidden');
  }

  witnessInput(body) {
    this.authorize(body);
    if (body.sourceId !== this.saved.sourceId || body.sourceGeneration !== this.saved.sourceGeneration) throw new Error('forbidden');
    if (canonical(body.target) !== canonical(this.saved.target)) throw new Error('stale_generation');
    this.checkCurrent();
  }

  checkCurrent() {
    if (!this.authorized) throw new Error('forbidden');
    if (canonical(this.currentTarget) !== canonical(this.saved.target)) throw new Error('stale_generation');
    if (this.currentSourceId !== this.saved.sourceId || this.currentSourceGeneration !== this.saved.sourceGeneration) throw new Error('stale_generation');
  }

  checkNewEpoch(body) {
    if (this.retired || this.nowMs >= this.epochExpiresAtMs || body.request.operationEpoch !== request.operationEpoch) throw new Error('receipt_expired');
  }

  operationKey(body) {
    return canonical(['app-1', 'alice', this.saved.bindingId, 'material-response', body.request.operationEpoch, body.request.requestId]);
  }

  queryResponse(body) {
    this.authorize(body);
    const previous = this.receipts.get(this.operationKey(body));
    if (!previous) throw new Error('receipt_expired');
    return copy(previous.receipt);
  }

  pending() {
    if (this.bindingState !== 'active') throw new Error('binding_closed');
    if (this.nowMs >= this.deadlineMs) throw new Error('request_expired');
    if (this.state !== 'pending') throw new Error('material_rejected');
  }

  upload(body) {
    this.witnessInput(body); this.pending();
    const witness = this.witnesses.get(body.artifactId);
    if (!witness) throw new Error('material_rejected');
    const data = Buffer.from(body.base64, 'base64');
    if (data.toString('base64') !== body.base64 || data.length !== body.bytes || hash(data) !== body.chunkSha256) throw new Error('material_rejected');
    const expectedBytes = Math.min(this.chunkBytes, witness.bytes - body.offset);
    if (!Number.isInteger(body.offset) || body.offset < 0 || body.offset % this.chunkBytes !== 0 ||
      (witness.bytes > 0 && body.offset >= witness.bytes) || expectedBytes !== body.bytes || body.bytes > this.chunkBytes) throw new Error('invalid_coverage');
    let upload = this.uploads.get(body.artifactId);
    if (!upload) {
      upload = { uploadId: 'upload-' + body.artifactId, artifact: witness, chunks: new Map(), state: 'receiving' };
      this.uploads.set(body.artifactId, upload);
    }
    const previous = upload.chunks.get(body.offset);
    if (previous) {
      if (!previous.equals(data)) throw new Error('request_id_conflict');
      return this.uploadStatus(body);
    }
    upload.chunks.set(body.offset, data); this.receivedBytes += data.length;
    const expectedCount = Math.max(1, Math.ceil(witness.bytes / this.chunkBytes));
    if (upload.chunks.size === expectedCount) {
      const full = Buffer.concat([...upload.chunks].sort(([a], [b]) => a - b).map(([, value]) => value));
      if (full.length !== witness.bytes || hash(full) !== witness.sha256) {
        this.state = 'rejected'; throw new Error('material_rejected');
      }
      upload.state = 'committed';
    }
    return this.uploadStatus(body);
  }

  uploadStatus(body) {
    this.authorize(body);
    if (this.nowMs >= this.deadlineMs) throw new Error('request_expired');
    const upload = this.uploads.get(body.artifactId);
    if (!upload) throw new Error('source_unavailable');
    return { uploadId: upload.uploadId, state: upload.state, artifact: copy(upload.artifact),
      receivedOffsets: [...upload.chunks.keys()].sort((a, b) => a - b),
      receivedBytes: [...upload.chunks.values()].reduce((sum, bytes) => sum + bytes.length, 0) };
  }

  respond(body) {
    this.authorize(body);
    const operationKey = this.operationKey(body);
    const semantic = copy(body); delete semantic.request;
    const digest = hash(canonical(semantic));
    const previous = this.receipts.get(operationKey);
    if (previous) {
      if (previous.digest !== digest) throw new Error('request_id_conflict');
      return copy(previous.receipt);
    }
    this.checkNewEpoch(body);
    this.witnessInput(body);
    this.pending();
    const seen = new Set();
    for (const result of body.results) {
      const requested = this.saved.requestedRecords.find((item) => item.recordId === result.recordId);
      if (!requested || requested.digest !== result.digest || seen.has(result.recordId)) throw new Error('material_rejected');
      seen.add(result.recordId);
      const actual = [result.payload, ...result.attachments].map((reference) => {
        const upload = [...this.uploads.values()].find((item) => item.uploadId === reference.uploadId);
        if (!upload || upload.state !== 'committed') throw new Error('material_rejected');
        return upload.artifact;
      });
      if (canonical(actual) !== canonical([requested.payload, ...requested.attachments])) throw new Error('material_rejected');
    }
    if (seen.size === 0) throw new Error('material_rejected');
    this.state = 'received'; this.revision++; this.coreQueued = true;
    const receipt = { state: this.state, revision: this.revision.toString(), acceptedRecordIds: [...seen] };
    this.receipts.set(operationKey, { digest, receipt });
    return copy(receipt);
  }

  terminate(reason) {
    if (reason === 'forbidden') this.authorized = false;
    if (reason === 'binding_closed') this.bindingState = 'closing';
    if (reason === 'request_expired' && this.state !== 'pending') return false;
    if (this.state === 'core-consumed') return false;
    this.state = 'rejected'; this.reason = reason; this.coreQueued = false; this.revision++;
    return true;
  }

  acquireCore() {
    // This synchronous reference step models one exclusion boundary with the lease increment.
    this.checkCurrent();
    if (!this.authorized || this.bindingState !== 'active' || !['received', 'verified'].includes(this.state)) throw new Error('material_rejected');
    this.coreQueued = false; this.coreHolds++;
  }

  releaseCore() { assert.ok(this.coreHolds > 0); this.coreHolds--; }
  canRelease() { return this.acknowledged && !this.coreQueued && this.coreHolds === 0 && ['core-consumed', 'rejected'].includes(this.state); }
}

test('SDK2-01b: response versus timeout, revoke and close uses one request CAS and core leases block release', () => {
  for (const reason of ['request_expired', 'forbidden', 'binding_closed']) {
    const first = new MaterialContractModel(); first.upload(uploadChunk);
    first.terminate(reason);
    assert.throws(() => first.respond(materialResponse), /material_rejected|forbidden|binding_closed/);
    assert.equal(first.receipts.size, 0, 'losing response did not commit a receipt');
    const responseFirst = new MaterialContractModel(); responseFirst.upload(uploadChunk);
    const receipt = responseFirst.respond(materialResponse);
    responseFirst.acknowledged = true;
    assert.equal(responseFirst.canRelease(), false, 'received/core-queued is not releasable');
    responseFirst.acquireCore();
    responseFirst.terminate(reason);
    assert.equal(responseFirst.canRelease(), false, 'timeout/revoke/close cannot free bytes held by core');
    assert.deepEqual([...responseFirst.receipts.values()][0].receipt, receipt, 'late termination does not rewrite admitted response history');
    responseFirst.releaseCore();
    if (reason === 'request_expired') {
      assert.equal(responseFirst.state, 'received', 'intake timeout cannot undo accepted CAS');
      responseFirst.state = 'core-consumed';
    }
    assert.equal(responseFirst.canRelease(), true);
    assert.equal(responseFirst.revision, reason === 'request_expired' ? 1n : 2n);
  }
  const drift = new MaterialContractModel(); drift.upload(uploadChunk);
  drift.currentTarget.generations.deletionGeneration = '1';
  assert.throws(() => drift.respond(materialResponse), /stale_generation/);
  assert.equal(drift.state, 'pending');
});
